void startseite() {
  server.send(200, "text/html", seite);
}

void zustandSenden() {
  if (digitalRead(led) == HIGH) {
    server.send(200, "text/plain", "an");
  } else {
    server.send(200, "text/plain", "aus");
  }
}

void ledAn() {
  digitalWrite(led, HIGH);
  Serial.println("Web: LED an");
  zustandSenden();
}

void ledAus() {
  digitalWrite(led, LOW);
  Serial.println("Web: LED aus");
  zustandSenden();
}

void temperaturSenden() {
  server.send(200, "text/plain", String(temperatur, 1));
}

void setup() {
  pinMode(led, OUTPUT);
  pinMode(taster, INPUT_PULLUP);
  Serial.begin(115200);
  delay(1000);

  ruhe = digitalRead(taster);
  tasterAlt = ruhe;

  sensoren.begin();
  sensoren.setWaitForConversion(false);
  sensoren.requestTemperatures();

  WiFi.softAP("ESP32-LED", "12345678");
  Serial.print("Webseite: http://");
  Serial.println(WiFi.softAPIP());

  server.on("/", startseite);
  server.on("/an", ledAn);
  server.on("/aus", ledAus);
  server.on("/status", zustandSenden);
  server.on("/temperatur", temperaturSenden);
  server.begin();
}

void loop() {
  server.handleClient();

  if (millis() - letzteMessung >= 1000) {
    letzteMessung = millis();
    temperatur = sensoren.getTempCByIndex(0);
    sensoren.requestTemperatures();
  }

  int tasterWert = digitalRead(taster);
  if (tasterWert != tasterAlt) {
    delay(20);
    tasterAlt = tasterWert;
    if (tasterWert != ruhe) {
      digitalWrite(led, HIGH);
      Serial.println("Taster: LED an");
    } else {
      digitalWrite(led, LOW);
      Serial.println("Taster: LED aus");
    }
  }
}
